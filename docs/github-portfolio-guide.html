3.2. Exponential Backoff & Jitter Algoritması (TypeScript)

Bu algoritma, başarısız olan isteklerin hedef sunucuyu çökertmeden akıllıca yeniden denenmesini sağlar:

// Exponential Backoff with Full Jitter
export function calculateRetryDelay(attempt: number, baseDelayMs = 1000, maxDelayMs = 3600000): number {
    // 2^attempt * baseDelay
    const exponentialBackoff = Math.min(maxDelayMs, baseDelayMs * Math.pow(2, attempt));
    // Full Jitter: 0 ile hesaplanan değer arasında rastgele bir süre (thundering herd koruması)
    return Math.floor(Math.random() * exponentialBackoff);
}

export async function handleWebhookRelay(payload: any, targetUrl: string, attempt = 0) {
    try {
        const response = await fetch(targetUrl, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload),
            signal: AbortSignal.timeout(5000) // 5 sn timeout
        });

        if (!response.ok) throw new Error(`HTTP_${response.status}`);
        return { status: 'DELIVERED' };
    } catch (err: any) {
        if (attempt < 5) {
            const delay = calculateRetryDelay(attempt);
            // Upstash Redis zset veya kuyruğuna planlanmış yeniden deneme ekle
            await scheduleRetry(payload, targetUrl, attempt + 1, Date.now() + delay);
            return { status: 'QUEUED_FOR_RETRY', nextAttemptInMs: delay };
        } else {
            // 5 deneme de başarısızsa Dead-Letter Queue'ya (DLQ) fırlat ve alarm çal
            await pushToDeadLetterQueue(payload, targetUrl, err.message);
            await triggerSlackAlert(targetUrl, err.message);
            return { status: 'DEAD_LETTER' };
        }
    }
}

// Normalizer: OpenAI -> Anthropic Messages API
export function transformOpenAIToAnthropic(openAIBody: any) {
    const systemMessage = openAIBody.messages.find((m: any) => m.role === 'system');
    const userAndAssistantMessages = openAIBody.messages
        .filter((m: any) => m.role !== 'system')
        .map((m: any) => ({
            role: m.role === 'user' ? 'user' : 'assistant',
            content: m.content
        }));

    return {
        model: 'claude-3-5-sonnet-20241022',
        max_tokens: openAIBody.max_tokens || 1024,
        system: systemMessage ? systemMessage.content : undefined,
        messages: userAndAssistantMessages,
        stream: openAIBody.stream ?? true
    };
}

4.3. Bütçe Kilidi ve Runaway Protection (Sonsuz Döngü Koruması)

Hard Cost Cap Mantığı: Redis üzerinde INCRBYFLOAT org:{id}:cost {tokenCost} komutu ile her çağrıda harcanan tutar atomik olarak toplanır. Kullanıcının belirlediği limit (örneğin günlük 20$) aşıldığı mikrosaniyede, sağlayıcıya gitmeden anında 429 Budget Exceeded cevabı döndürülür. Fatura asla patlamaz.

5.2. Çekirdek Kriptografi Motoru (Node.js Crypto)

import crypto from 'node:crypto';

const ALGORITHM = 'aes-256-gcm';
const IV_LENGTH = 16;
const SALT_LENGTH = 32;

export function encryptEnv(plainText: string, masterKey: string): string {
    const salt = crypto.randomBytes(SALT_LENGTH);
    const key = crypto.pbkdf2Sync(masterKey, salt, 100000, 32, 'sha512');
    const iv = crypto.randomBytes(IV_LENGTH);

    const cipher = crypto.createCipheriv(ALGORITHM, key, iv);
    const encrypted = Buffer.concat([cipher.update(plainText, 'utf8'), cipher.final()]);
    const tag = cipher.getAuthTag();

    // Format: [Salt (32)][IV (16)][Auth Tag (16)][Encrypted Data]
    return Buffer.concat([salt, iv, tag, encrypted]).toString('base64');
}

export function decryptEnv(cipherPayloadBase64: string, masterKey: string): string {
    const buffer = Buffer.from(cipherPayloadBase64, 'base64');
    const salt = buffer.subarray(0, 32);
    const iv = buffer.subarray(32, 48);
    const tag = buffer.subarray(48, 64);
    const encryptedData = buffer.subarray(64);

    const key = crypto.pbkdf2Sync(masterKey, salt, 100000, 32, 'sha512');
    const decipher = crypto.createDecipheriv(ALGORITHM, key, iv);
    decipher.setAuthTag(tag);

    return decipher.update(encryptedData) + decipher.final('utf8');
}
